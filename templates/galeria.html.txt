        // Transição SPA com efeito Fade-In
        function abrirAlbum(pastaId) {
            const secaoPastas = document.getElementById('secao-pastas');
            const albumAlvo = document.getElementById('album-' + pastaId);

            secaoPastas.classList.remove('fade-in-active');
            
            setTimeout(() => {
                secaoPastas.classList.remove('visivel');
                
                if (albumAlvo) {
                    albumAlvo.classList.add('visivel');
                    setTimeout(() => {
                        albumAlvo.classList.add('fade-in-active');
                    }, 50);
                }
                window.scrollTo({ top: 0, behavior: 'smooth' });
            }, 300);
        }

        function voltarParaPastas() {
            const secoes = document.querySelectorAll('.secao-album');
            const secaoPastas = document.getElementById('secao-pastas');

            secoes.forEach(secao => secao.classList.remove('fade-in-active'));

            setTimeout(() => {
                secoes.forEach(secao => secao.classList.remove('visivel'));
                
                secaoPastas.classList.add('visivel');
                setTimeout(() => {
                    secaoPastas.classList.add('fade-in-active');
                }, 50);
                window.scrollTo({ top: 0, behavior: 'smooth' });
            }, 300);
        }
    </script>
